from datetime import datetime from fastapi import FastAPI, HTTPException, Depends from fastapi.responses import HTMLResponse from pydantic import BaseModel from sqlalchemy import create_engine, Column, Integer, String from sqlalchemy.ext.declarative import declarative_base from sqlalchemy.orm import sessionmaker, Session app = FastAPI() # База данных SQLite для хранения данных пользователей SQLALCHEMY_DATABASE_URL = "sqlite:///./milk_app.db" engine = create_engine(SQLALCHEMY_DATABASE_URL, connect_args={"check_same_thread": False}) SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine) Base = declarative_base() class UserDB(Base): __tablename__ = "users" telegram_id = Column(String, primary_key=True, index=True) milk_balance = Column(Integer, default=0) streak_day = Column(Integer, default=1) last_claim_date = Column(String, nullable=True) wallet_address = Column(String, nullable=True) Base.metadata.create_all(bind=engine) def get_db(): db = SessionLocal() try: yield db finally: db.close() # Уютная таблица наград за 7 дней STREAK_REWARDS = {1: 5, 2: 10, 3: 20, 4: 30, 5: 40, 6: 50, 7: 70} class UserActionRequest(BaseModel): telegram_id: str @app.post("/api/get_user") def get_or_create_user(data: UserActionRequest, db: Session = Depends(get_db)): user = db.query(UserDB).filter(UserDB.telegram_id == data.telegram_id).first() if not user: user = UserDB(telegram_id=data.telegram_id, milk_balance=0, streak_day=1) db.add(user) db.commit() db.refresh(user) return { "telegram_id": user.telegram_id, "milk_balance": user.milk_balance, "streak_day": user.streak_day, "wallet_address": user.wallet_address } @app.post("/api/daily_reward") def claim_daily_reward(data: UserActionRequest, db: Session = Depends(get_db)): user = db.query(UserDB).filter(UserDB.telegram_id == data.telegram_id).first() if not user: raise HTTPException(status_code=404, detail="Пользователь не найден") today = datetime.utcnow().strftime("%Y-%m-%d") if user.last_claim_date == today: raise HTTPException(status_code=400, detail="Награда сегодня уже получена!") # Если пропустили день (больше 1 дня разницы), сбрасываем серию на 1-й день if user.last_claim_date: last_date = datetime.strptime(user.last_claim_date, "%Y-%m-%d") if (datetime.utcnow() - last_date).days > 1: user.streak_day = 1 reward = STREAK_REWARDS.get(user.streak_day, 5) user.milk_balance += reward user.last_claim_date = today current_streak = user.streak_day # Цикл серии от 1 до 7 дней if user.streak_day < 7: user.streak_day += 1 else: user.streak_day = 1 db.commit() return {"success": True, "reward": reward, "new_balance": user.milk_balance, "next_streak": user.streak_day} @app.post("/api/watch_ad") def watch_ad(data: UserActionRequest, db: Session = Depends(get_db)): user = db.query(UserDB).filter(UserDB.telegram_id == data.telegram_id).first() if not user: raise HTTPException(status_code=404, detail="Пользователь не найден") user.milk_balance += 5 # +5 молока за просмотр рекламы db.commit() return {"success": True, "added": 5, "new_balance": user.milk_balance} @app.get("/api/exchange_rates") def get_rates(): return { "milk_to_stars": 100, # 100 молока = 1 Star ⭐ "milk_to_gram": 50 # 50 молока = 1 GRAM 💎 } # Отдаем красивый фронтенд прямо с сервера @app.get("/", response_class=HTMLResponse) def serve_frontend(): return HTML_CONTENT HTML_CONTENT = """ Milk Mini App

"""
