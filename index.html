import time # Таблица характеристик уровней COW_LEVELS = { 1: {"income": 20}, 2: {"income": 45}, 3: {"income": 100}, 4: {"income": 220}, 5: {"income": 480}, } MAX_HOURS = 4 # Лимит накопления в часах def claim_milk(user_data): current_time = time.time() last_time = user_data["last_claim_time"] level = user_data["cow_level"] # Прошло времени с последнего сбора (в секундах) elapsed_seconds = current_time - last_time max_seconds = MAX_HOURS * 3600 # 4 часа в секундах # Ограничиваем время максимумом в 4 часа (если прошло больше, молоко больше не фармилось) effective_seconds = min(elapsed_seconds, max_seconds) if effective_seconds < 60: # например, даем собирать хотя бы раз в минуту или проверяем по факту return "Еще рано собирать молоко!" # Считаем, сколько циклов по 4 часа прошло, ИЛИ просто выдаем фиксированно за 4 часа, # если прошло больше 4 часов. # Так как фарм останавливается через 4 часа, за один сбор игрок получает ровно порцию за 4 часа: earned_milk = COW_LEVELS[level]["income"] # Обновляем баланс и время последнего сбора user_data["milk_balance"] += earned_milk user_data["last_claim_time"] = current_time # Сбрасываем таймер return f"Вы успешно собрали {earned_milk} молока!"
